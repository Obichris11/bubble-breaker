# Plan 02 — Game Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the complete Standard-style Bubble Breaker rules as a pure, immutable, deterministic TypeScript engine with unit and property tests at ≥ 95 % coverage.

**Architecture:** `src/app/engine/` contains only framework-free TypeScript (no Angular, no DOM, lint-enforced since Plan 01).
- A `Game` holds the seed, rules, a colour per `BallId`, an immutable row-major `Board` and the score.
- `burst()` returns a `MoveResult` with the intermediate boards (after gravity, after collapse), which the state layer (Plan 03) uses to drive animations.
- Every rule invariant from `01-rules.md` becomes a fast-check property.

**Tech Stack:** TypeScript 6.0 (strict + `noUncheckedIndexedAccess` + `exactOptionalPropertyTypes`), Vitest 5 via `ng test`, fast-check 4.

**Spec:** [2026-09-27-bubble-breaker-rewrite-design.md](../specs/2026-09-27-bubble-breaker-rewrite-design.md) · rules in [01-rules.md](../../01-rules.md) · engine API in [03-architecture.md](../../03-architecture.md#engine-pure-typescript)

## Global Constraints

- Plan 01 is done: `main` exists and is protected; `npm run check` and `npm run test:ci` work.
- Engine files import nothing outside `src/app/engine/`. No `@angular/*`, no `rxjs`, no DOM APIs.
- Nothing is mutated. Every function returns new values, and all public types are `readonly`.
- Board: 11 columns × 12 rows. 5 colours (0 red, 1 blue, 2 green, 3 yellow, 4 purple). Orthogonal adjacency. Minimum group size 2.
- Score per burst = `X(X − 1)`. No end bonus. No undo.
- Gravity pulls balls down (row 0 is the top). Empty columns close up to the **right**.
- RNG: mulberry32. The seed is normalised with `>>> 0`.
- Coverage for `src/app/engine/**` ≥ 95 % lines, statements, functions and branches. `testing/` is excluded.
- Lint bans `any` and non-null assertions in `src/` (allowed in specs and `testing/`), and default exports.
- Test files use Vitest globals (`describe`, `it`, `expect`), like the Angular-generated specs.

```ts
/**
 * mulberry32: small, fast, seedable PRNG (rule R5). Identical to the design
 * prototype so seeded boards match docs/design/Game Screen.dc.html.
 */
export function createRng(seed: number): () => number {
  let state = seed | 0;
  return () => {
    state = (state + 0x6d2b79f5) | 0;
    let t = Math.imul(state ^ (state >>> 15), 1 | state);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

```ts
/** Stable identity of one ball for the whole game (used for animation tracking). */
export type BallId = number;

export interface Cell {
  readonly row: number;
  readonly col: number;
}

/** Immutable grid. Row 0 is the top row; `cells` is row-major (index = row·columns + col). */
export interface Board {
  readonly columns: number;
  readonly rows: number;
  readonly cells: readonly (BallId | null)[];
}

/** Index access that fails loudly instead of returning `undefined`. */
export function itemAt<T>(items: readonly T[], index: number): T {
  const item = items[index];
  if (item === undefined) {
    throw new RangeError(`Index ${String(index)} is out of range (length ${String(items.length)})`);
  }
  return item;
}

export function isInside(board: Board, cell: Cell): boolean {
  return (
    Number.isInteger(cell.row) &&
    Number.isInteger(cell.col) &&
    cell.row >= 0 &&
    cell.row < board.rows &&
    cell.col >= 0 &&
    cell.col < board.columns
  );
}

export function ballAt(board: Board, cell: Cell): BallId | null {
  if (!isInside(board, cell)) {
    throw new RangeError(
      `Cell (${String(cell.row)}, ${String(cell.col)}) is outside the ${String(board.rows)}×${String(board.columns)} board`,
    );
  }
  return itemAt(board.cells, cell.row * board.columns + cell.col);
}

export function countBalls(board: Board): number {
  return board.cells.filter((id) => id !== null).length;
}

/** Columns left to right, each ordered top to bottom. */
export function columnsOf(board: Board): (BallId | null)[][] {
  return Array.from({ length: board.columns }, (_, col) =>
    board.cells.filter((_, index) => index % board.columns === col),
  );
}

export function fromColumns(columns: readonly (readonly (BallId | null)[])[], rows: number): Board {
  const width = columns.length;
  const cells = new Array<BallId | null>(width * rows).fill(null);
  columns.forEach((column, col) => {
    column.forEach((id, row) => {
      cells[row * width + col] = id;
    });
  });
  return { columns: width, rows, cells };
}
```

```ts
import type { BallId, Board, Cell } from '../board';
import { colorOf, type Game } from '../game';
import { STANDARD_RULES, type Color } from '../rules';

const LETTERS = 'rbgyp';

/**
 * Builds a Game from rows of colour letters (r b g y p) and '.' for empty cells.
 * Ball ids are assigned row-major to non-empty cells, starting at 0.
 */
export function gameFrom(rows: readonly string[], score = 0): Game {
  const columns = rows[0]?.length ?? 0;
  const colors: Color[] = [];
  const cells: (BallId | null)[] = [];
  for (const line of rows) {
    if (line.length !== columns) {
      throw new Error(`All rows must have ${String(columns)} cells: "${line}"`);
    }
    for (const letter of line) {
      if (letter === '.') {
        cells.push(null);
        continue;
      }
      const color = LETTERS.indexOf(letter);
      if (color < 0) {
        throw new Error(`Unknown colour letter "${letter}"`);
      }
      cells.push(colors.length);
      colors.push(color as Color);
    }
  }
  return {
    seed: 0,
    rules: { ...STANDARD_RULES, columns, rows: rows.length },
    colors,
    board: { columns, rows: rows.length, cells },
    score,
  };
}

/** Renders a board as letter rows using the game's colours. */
export function lettersOf(game: Game, board: Board = game.board): string[] {
  return Array.from({ length: board.rows }, (_, row) =>
    board.cells
      .slice(row * board.columns, (row + 1) * board.columns)
      .map((id) => (id === null ? '.' : LETTERS.charAt(colorOf(game, id))))
      .join(''),
  );
}

/** Sorts cells top-to-bottom, left-to-right, so groups compare independent of fill order. */
export function sortCells(cells: readonly Cell[]): Cell[] {
  return [...cells].sort((a, b) => a.row - b.row || a.col - b.col);
}
```

```ts
import { itemAt, type BallId, type Board } from './board';
import { createRng } from './rng';
import { STANDARD_RULES, type Color, type Rules } from './rules';

export interface Game {
  /** Normalised uint32 seed the board was dealt from. */
  readonly seed: number;
  readonly rules: Rules;
  /** Colour of each ball, indexed by BallId. Never changes during a game. */
  readonly colors: readonly Color[];
  readonly board: Board;
  readonly score: number;
}

/** Deals a full board: every cell gets an independent uniform-random colour (rules R1, R4, R5). */
export function newGame(seed: number, rules: Rules = STANDARD_RULES): Game {
  const normalizedSeed = seed >>> 0;
  const random = createRng(normalizedSeed);
  const size = rules.columns * rules.rows;
  const colors = Array.from({ length: size }, () => Math.floor(random() * rules.colors) as Color);
  const cells = Array.from({ length: size }, (_, id) => id);
  return {
    seed: normalizedSeed,
    rules,
    colors,
    board: { columns: rules.columns, rows: rules.rows, cells },
    score: 0,
  };
}

export function colorOf(game: Game, id: BallId): Color {
  return itemAt(game.colors, id);
}
```

```ts
import { columnsOf, fromColumns, type BallId, type Board } from './board';

/** Rule R11: balls above removed cells fall straight down. */
export function applyGravity(board: Board): Board {
  const settled = columnsOf(board).map((column) => {
    const balls = column.filter((id): id is BallId => id !== null);
    return [...new Array<null>(board.rows - balls.length).fill(null), ...balls];
  });
  return fromColumns(settled, board.rows);
}

/**
 * Rule R12: empty columns close up to the right, so empty space collects on the left.
 * Expects gravity to have been applied. Returns `board` itself when no column moves.
 */
export function collapseRight(board: Board): Board {
  const columns = columnsOf(board);
  const isEmpty = columns.map((column) => column.every((id) => id === null));
  const firstFilled = isEmpty.indexOf(false);
  const alreadyCollapsed = firstFilled === -1 || isEmpty.slice(firstFilled).every((empty) => !empty);
  if (alreadyCollapsed) {
    return board;
  }
  const filled = columns.filter((_, col) => !isEmpty[col]);
  const empties = Array.from({ length: columns.length - filled.length }, () =>
    new Array<null>(board.rows).fill(null),
  );
  return fromColumns([...empties, ...filled], board.rows);
}
```

```ts
import { ballAt, isInside, type Cell } from './board';
import { colorOf, type Game } from './game';

const NEIGHBOURS: readonly Cell[] = [
  { row: -1, col: 0 },
  { row: 1, col: 0 },
  { row: 0, col: -1 },
  { row: 0, col: 1 },
];

/** Breadth-first flood fill over orthogonal neighbours of the same colour (rule R6). */
function floodFill(game: Game, start: Cell): Cell[] {
  const startId = ballAt(game.board, start);
  if (startId === null) {
    return [];
  }
  const color = colorOf(game, startId);
  const keyOf = (cell: Cell): number => cell.row * game.board.columns + cell.col;
  const seen = new Set<number>([keyOf(start)]);
  const queue: Cell[] = [start];
  const group: Cell[] = [];
  for (let current = queue.shift(); current !== undefined; current = queue.shift()) {
    group.push(current);
    for (const step of NEIGHBOURS) {
      const next = { row: current.row + step.row, col: current.col + step.col };
      if (!isInside(game.board, next) || seen.has(keyOf(next))) {
        continue;
      }
      const id = ballAt(game.board, next);
      if (id !== null && colorOf(game, id) === color) {
        seen.add(keyOf(next));
        queue.push(next);
      }
    }
  }
  return group;
}

/** Rules R6/R7: the group at `cell`, or [] if the cell is empty or the group is too small. */
export function groupAt(game: Game, cell: Cell): readonly Cell[] {
  const group = floodFill(game, cell);
  return group.length >= game.rules.minGroup ? group : [];
}

export function allGroups(game: Game): readonly (readonly Cell[])[] {
  const { columns, rows } = game.board;
  const visited = new Set<number>();
  const groups: Cell[][] = [];
  for (let row = 0; row < rows; row++) {
    for (let col = 0; col < columns; col++) {
      if (visited.has(row * columns + col)) {
        continue;
      }
      const group = floodFill(game, { row, col });
      group.forEach((cell) => visited.add(cell.row * columns + cell.col));
      if (group.length >= game.rules.minGroup) {
        groups.push(group);
      }
    }
  }
  return groups;
}

/** Rule R13: the game is over when no group of `minGroup` or more remains. */
export function hasMoves(game: Game): boolean {
  return allGroups(game).length > 0;
}
```

```ts
import { ballAt, type BallId, type Board, type Cell } from './board';
import type { Game } from './game';
import { groupAt, hasMoves } from './groups';
import { applyGravity, collapseRight } from './physics';
import { scoreFor } from './rules';

export interface MoveResult {
  readonly removed: readonly BallId[];
  readonly points: number;
  readonly afterGravity: Board;
  /** Same reference as `afterGravity` when no column slid. */
  readonly afterCollapse: Board;
  readonly game: Game;
  readonly gameOver: boolean;
}

/** Bursts the group at `cell` (rules R8–R13). Returns null when there is nothing to burst. */
export function burst(game: Game, cell: Cell): MoveResult | null {
  const group = groupAt(game, cell);
  if (group.length === 0) {
    return null;
  }
  const removed = group
    .map((member) => ballAt(game.board, member))
    .filter((id): id is BallId => id !== null);
  const removedSet = new Set(removed);
  const cleared: Board = {
    ...game.board,
    cells: game.board.cells.map((id) => (id !== null && removedSet.has(id) ? null : id)),
  };
  const afterGravity = applyGravity(cleared);
  const afterCollapse = collapseRight(afterGravity);
  const points = scoreFor(removed.length);
  const next: Game = { ...game, board: afterCollapse, score: game.score + points };
  return { removed, points, afterGravity, afterCollapse, game: next, gameOver: !hasMoves(next) };
}
```

```ts
interface Step {
  readonly before: Game;
  readonly result: MoveResult;
}

/** Plays up to `choices.length` moves, choosing among the available groups by index. */
function play(seed: number, choices: readonly number[]): { game: Game; steps: Step[] } {
  let game = newGame(seed);
  const steps: Step[] = [];
  for (const choice of choices) {
    const groups = allGroups(game);
    if (groups.length === 0) break;
    const group = itemAt(groups, choice % groups.length);
    const result = burst(game, itemAt(group, 0));
    if (result === null) throw new Error('allGroups returned a group that cannot burst');
    steps.push({ before: game, result });
    game = result.game;
  }
  return { game, steps };
}

function hasFloatingBall(board: Board): boolean {
  for (let col = 0; col < board.columns; col++) {
    let seenBall = false;
    for (let row = 0; row < board.rows; row++) {
      const id = itemAt(board.cells, row * board.columns + col);
      if (id !== null) seenBall = true;
      else if (seenBall) return true;
    }
  }
  return false;
}

function emptyColumnsAreOnTheLeft(board: Board): boolean {
  let seenFilled = false;
  for (let col = 0; col < board.columns; col++) {
    const filled = Array.from({ length: board.rows }, (_, row) =>
      itemAt(board.cells, row * board.columns + col),
    ).some((id) => id !== null);
    if (filled) seenFilled = true;
    else if (seenFilled) return false;
  }
  return true;
}

/** Independent game-over oracle: any two orthogonally adjacent balls with the same colour. */
function hasAdjacentPair(game: Game): boolean {
  const { columns, rows, cells } = game.board;
  for (let row = 0; row < rows; row++) {
    for (let col = 0; col < columns; col++) {
      const id = itemAt(cells, row * columns + col);
      if (id === null) continue;
      const right = col + 1 < columns ? itemAt(cells, row * columns + col + 1) : null;
      const below = row + 1 < rows ? itemAt(cells, (row + 1) * columns + col) : null;
      if (right !== null && colorOf(game, right) === colorOf(game, id)) return true;
      if (below !== null && colorOf(game, below) === colorOf(game, id)) return true;
    }
  }
  return false;
}
```
